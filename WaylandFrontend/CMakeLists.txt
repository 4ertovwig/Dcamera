set(LIB_NAME wcamera)

add_library(${LIB_NAME} STATIC)
set_target_properties(${LIB_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND_CLIENT REQUIRED wayland-client)
pkg_check_modules(WAYLAND_CURSOR REQUIRED wayland-cursor)
pkg_check_modules(XKBCOMMON REQUIRED xkbcommon)

find_program(WAYLAND_SCANNER NAMES wayland-scanner)
if(NOT WAYLAND_SCANNER)
    message(FATAL_ERROR "wayland-scanner not found")
endif()

find_package(Threads REQUIRED)

# find xdg-shell.xml
find_path(WAYLAND_PROTOCOLS_DIR
    NAMES stable/xdg-shell/xdg-shell.xml
    PATHS
        /usr/share/wayland-protocols
        /usr/local/share/wayland-protocols
    REQUIRED
)

set(XDG_SHELL_XML "${WAYLAND_PROTOCOLS_DIR}/stable/xdg-shell/xdg-shell.xml")
set(DMABUF_XML "${WAYLAND_PROTOCOLS_DIR}/stable/linux-dmabuf/linux-dmabuf-v1.xml")
set(VIEWPORTER_XML "${WAYLAND_PROTOCOLS_DIR}/stable/viewporter/viewporter.xml")

if(NOT EXISTS ${XDG_SHELL_XML})
    message(FATAL_ERROR "xdg-shell.xml not found at: ${XDG_SHELL_XML}")
endif()

if(NOT EXISTS ${DMABUF_XML})
    message(FATAL_ERROR "linux-dmabuf-v1.xml not found at: ${DMABUF_XML}")
endif ()

if(NOT EXISTS ${VIEWPORTER_XML})
    message(FATAL_ERROR "viewporter.xml not found at: ${DMABUFVIEWPORTER_XML_XML}")
endif ()

message(STATUS "Using protocol XML: ${XDG_SHELL_XML}")

# Generate client *.h
add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.h
    COMMAND ${WAYLAND_SCANNER}
        client-header
        ${XDG_SHELL_XML}
        ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.h
    DEPENDS ${XDG_SHELL_XML}
    COMMENT "Generating xdg-shell client protocol header"
    VERBATIM
)

# Generate client .c
add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.c
    COMMAND ${WAYLAND_SCANNER}
        private-code
        ${XDG_SHELL_XML}
        ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.c
    DEPENDS ${XDG_SHELL_XML}
    COMMENT "Generating xdg-shell client protocol code"
    VERBATIM
)

# Generate linux-dmabuf-unstable-v1-client-protocol.h
add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/linux-dmabuf-unstable-v1-client-protocol.h
           ${CMAKE_CURRENT_BINARY_DIR}/linux-dmabuf-unstable-v1-client-protocol.c
    COMMAND ${WAYLAND_SCANNER}
        private-code
        ${DMABUF_XML}
        ${CMAKE_CURRENT_BINARY_DIR}/linux-dmabuf-unstable-v1-client-protocol.h
        COMMAND ${WAYLAND_SCANNER}
        private-code
        ${DMABUF_XML}
        ${CMAKE_CURRENT_BINARY_DIR}/linux-dmabuf-unstable-v1-client-protocol.c
    DEPENDS ${DMABUF_XML}
    COMMENT "Generating dma-buf support client protocol code"
    VERBATIM
)
    
# Generate viewporter-client-protocol.h
add_custom_command(
    OUTPUT 
        ${CMAKE_CURRENT_BINARY_DIR}/viewporter-client-protocol.h
        ${CMAKE_CURRENT_BINARY_DIR}/viewporter-client-protocol.c
    COMMAND ${WAYLAND_SCANNER} client-header
        ${VIEWPORTER_XML} viewporter-client-protocol.h
    COMMAND ${WAYLAND_SCANNER} private-code
        ${VIEWPORTER_XML} viewporter-client-protocol.c
    DEPENDS ${VIEWPORTER_XML}
    COMMENT "Generating viewporter protocol"
)

# generate_protocols extra aim
add_custom_target(generate_protocols ALL
    DEPENDS
        ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.h
        ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.c
        ${CMAKE_CURRENT_BINARY_DIR}/linux-dmabuf-unstable-v1-client-protocol.h
        ${CMAKE_CURRENT_BINARY_DIR}/viewporter-client-protocol.h
)

add_dependencies(${LIB_NAME} generate_protocols)

set(SOURCES
    videostream.c
    audiostream.c
    main.c
    xdg-shell-client-protocol.c                     #autogenerated
    viewporter-client-protocol.c                    #autogenerated
    # may be later
    # linux-dmabuf-unstable-v1-client-protocol.c    #autogenerated
    #wl_functions.c
    wl_common.c
    wl_display.c
    wl_functions.c
    wl_handle.c
    wl_keyboard.c
    wl_pointer.c
    wl_window.c
    xkb_common.c)

target_sources(${LIB_NAME} PRIVATE ${SOURCES})

# main gui library
target_link_libraries(${LIB_NAME}
    PRIVATE
    ${WAYLAND_CLIENT_LIBRARIES}
    ${WAYLAND_CURSOR_LIBRARIES}
    ${XKBCOMMON_LIBRARIES}
    Threads::Threads
)

# Directory include
target_include_directories(${LIB_NAME} PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}
    ${WAYLAND_CLIENT_INCLUDE_DIRS}
    ${WAYLAND_CURSOR_INCLUDE_DIRS}
    ${XKBCOMMON_INCLUDE_DIRS}
)

target_compile_options(${LIB_NAME} PRIVATE 
    ${WAYLAND_CLIENT_CFLAGS}
)

# link video streams
if (USE_VIDEO_BACKEND_V4L2)
    target_link_libraries(${LIB_NAME}
        PRIVATE
        video_stream
    )
elseif (USE_VIDEO_BACKEND_UVC)
    target_link_libraries(${LIB_NAME}
        PRIVATE
        ${LIBUSB_LIBRARIES}
        ${LIBUVC_LIBRARIES}
        video_stream
    )
endif ()

# link audio streams
if (USE_AUDIO_BACKEND_PULSE OR USE_AUDIO_BACKEND_PIPEWIRE OR USE_AUDIO_BACKEND_ALSA)
    target_link_libraries(${LIB_NAME}
        PRIVATE
        audio_stream
    )
    include_directories(${PIPEWIRE_INCLUDE_DIRS})
endif ()
